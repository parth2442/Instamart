LAYOUT HIERARCHY (exactly like help.py):
─────────────────────────────────────────────────────────────────

    LayoutView (timeout=120 or None for persistent)
    │
    ├── Container (accent_color=color.black)          # ← main wrapper
    │     ├── Section                                  # ← title + thumbnail combo
    │     │     ├── TextDisplay(f"### Heading")
    │     │     ├── TextDisplay("description...")
    │     │     └── accessory=Thumbnail(
    │     │             media=discord.UnfurledMediaItem(url=...),
    │     │             description="alt text"
    │     │         )
    │     ├── Separator(spacing=discord.SeparatorSpacing.large)
    │     ├── TextDisplay("content with **markdown**")
    │     ├── Separator(...)
    │     ├── ActionRow                                  # ← button row
    │     │     ├── Button(label, style, emoji, url/custom_id)
    │     │     └── Button(...)
    │     └── TextDisplay("-# Footer text")
    │
    ├── ActionRow                                       # ← dropdown row
    │     └── Select(placeholder, options=[...])
    │
    └── ActionRow                                       # ← bottom buttons row
          ├── Button(...)
          └── Button(...)

from discord import ui
from discord.ext import commands
import discord
from instamart.style import color
from instamart.style import emoji as e


class BaseView(ui.LayoutView):
    def __init__(self, bot, ctx, timeout=120):
        super().__init__(timeout=timeout)
        self.bot = bot
        self.ctx = ctx
        self.message: discord.Message = None

    async def interaction_check(self, interaction: discord.Interaction) -> bool:
        if interaction.user.id != self.ctx.author.id:
            await interaction.response.send_message(
                embed=discord.Embed(
                    description="You are not allowed to use this interaction",
                    color=color.error,
                ),
                ephemeral=True,
                delete_after=10,
            )
            return False
        return True

    async def on_timeout(self):
        for item in self.children:
            item.disabled = True
        if self.message:
            try:
                await self.message.edit(view=self)
            except (discord.NotFound, discord.HTTPException):
                pass


class HomeView(BaseView):
